  /*
   * Internal FS_FILE layout (offsets confirmed from assembly):
   *   +0x00  void   *pFileObj      pointer to file object
   *   +0x08  I16     ErrCode       last error  (I16 = signed 16-bit)
   *   +0x0A  U8      Driver        driver index (0 = invalid)
   *   +0x0B  U8      Flags         bit 3 = opened-for-reading
   */
  #define FS_FILE_ACCESS_FLAG_R   (1u << 3)   /* 0x08 */
  #define FS_ERR_NOTOPEN_FOR_READ (-5)

  static U32 FS__Read(FS_FILE *pFile, void *pData, U32 NumBytes)
  {
      /* Guard: all inputs must be non-null/non-zero */
      if (NumBytes == 0u)       return 0u;
      if (pFile    == NULL)     return 0u;

      U8    driver  = *(U8  *)((U8 *)pFile + 0x0A);  /* pFile->Driver   */
      void *pFileObj = *(void **)pFile;               /* pFile->pFileObj */

      if (driver   == 0u)       return 0u;
      if (pFileObj == NULL)     return 0u;

      /* Permission check — was the file opened for reading? */
      U8 flags = *(U8 *)((U8 *)pFile + 0x0B);        /* pFile->Flags    */
      if ((flags & FS_FILE_ACCESS_FLAG_R) == 0u) {
          *(I16 *)((U8 *)pFile + 0x08) = FS_ERR_NOTOPEN_FOR_READ;
          return 0u;
      }

      /* All checks passed — dispatch to FAT layer */
      return FS_FAT_Read(pFile, pData, NumBytes);
  }

C EQUIVALENT OF THE POLLING LOOP:
  /*
   * Excerpt from _MMC_Read() — wait for SD card data-start token, then
   * read the sector. Called once per 512-byte sector needed.
   *
   * hw_unit    = SPI hardware channel index
   * pData      = destination buffer
   * maxRetries = configured timeout (e.g. 1000–65535 iterations)
   */

  static int _WaitAndReadSector(U8 hw_unit, U8 *pData, U32 maxRetries)
  {
      U8  token;
      U32 retryCount = 0;

      /* ── POLLING LOOP ───────────────────────────────────────────────────
       * The SD card holds MISO = 0xFF while its internal controller is
       * busy reading from flash cells. We must poll one byte at a time
       * until we see the data-start token 0xFE, then read the sector.
       *
       * CPU is 100% occupied here — no sleep, no yield, no DMA.
       * At 1 MHz SPI: each iteration takes ~8 µs (1 byte × 8 bits).
       * Typical SD card latency: 1–10 ms → 125–1250 iterations.
       * ─────────────────────────────────────────────────────────────────*/
      do {
          FS_MMC_HW_X_Read(hw_unit, &token, 1);   /* read 1 byte over SPI */

          if (token == 0xFEu) {
              goto token_received;                 /* data-start token found */
          }

          retryCount++;

      } while (retryCount < maxRetries);

      return -1;    /* timeout: SD card never sent the token */

  token_received:
      /* Read 512 bytes of sector payload — one call to your HAL */
      FS_MMC_HW_X_Read(hw_unit, pData, 512);      /* 0x200 bytes */

      /* Read 2-byte CRC16 that trails every sector on the SPI bus */
      U8 crcBuf[2];
      FS_MMC_HW_X_Read(hw_unit, crcBuf, 2);

      /* Verify CRC if enabled (0xFFFF = disabled sentinel) */
      U16 crcReceived = ((U16)crcBuf[0] << 8) | crcBuf[1];
      U16 crcComputed = _CalcDataCRC16(pData, 512);   /* T.53 in binary */
      if (crcComputed != 0xFFFFu && crcComputed != crcReceived) {
          return -1;   /* CRC mismatch */
      }

      return 0;   /* success */
  }

  1. FS_Read() has ZERO logic — it is a 4-byte tail-call to FS__Read().

  2. FS__Read() does only null/permission guards (46 bytes), then
     tail-calls FS_FAT_Read() for all actual work.

  3. The polling loop is inside _MMC_Read() (SPI driver layer), NOT
     inside FS_Read/FS__Read/FS_FAT_Read. It polls FS_MMC_HW_X_Read(1)
     in a tight loop until byte 0xFE (SD data-start token) is received.

  4. The CPU is 100% busy during:
       (a) the polling loop  — waiting for 0xFE  (~1–10 ms typical)
       (b) the 512-byte read — FS_MMC_HW_X_Read(512) — depends on YOUR HAL
     Whether (b) uses DMA is entirely determined by your FS_MMC_HW_X_Read
     implementation, not by emFile itself.

  5. _MMC_Read retries the entire command up to 3 times (sb register
     starts at 3, decremented on each failure) before returning -1.

  6. CRC16 checking is supported but disabled by default
     (sentinel value 0xFFFF from T.53 / _CalcDataCRC16Dummy).

  7. FS_FRead() calls FS__Read() directly (not FS_Read), multiplies
     ItemSize×NumItems before the call, divides bytesRead/ItemSize after.
